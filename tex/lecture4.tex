\begin{frame}{Where we are}
  In the simply typed $\lambda$-calculus, a term receives one type at a time.
  For example,
  \[
    \lambda x.\,x : A \to A
  \]
  for each chosen type $A$.

  \vfill

  This lecture asks what happens if a single program may be used uniformly at
  every type.

  \begin{center}
    \begin{tabular}{ll}
      Simply typed $\lambda$-calculus & one type at a time \\
      Polymorphic $\lambda$-calculus & one term for all types
    \end{tabular}
  \end{center}

  \vfill

  The technical object is System~F: simply typed $\lambda$-calculus extended
  with universal quantification over types.
\end{frame}

\section{Polymorphic \texorpdfstring{$\lambda$}{λ}-Calculus: Static}

\begin{frame}{Why polymorphism?}
  Many programs do not inspect the concrete type of their inputs.

  \begin{description}
    \item[Identity]
      \[
        \mathtt{id} : \forall X.\,X \to X
      \]
      should work at every type $X$.

    \item[First projection]
      \[
        \mathtt{proj}_1 : \forall X.\,\forall Y.\,X \to Y \to X
      \]
      should ignore the second argument uniformly.

    \item[Church numerals]
      \[
        \nat \defeq \forall X.\,(X \to X) \to X \to X
      \]
      should iterate any endofunction on any carrier type.
  \end{description}
\end{frame}

\begin{frame}{What changed from simply typed \(\lambda\)-calculus?}
  System~F adds two new ideas.

  \begin{center}
    \begin{tabular}{lll}
      & Introduction & Elimination \\[2pt]
      term functions
        & \(\lambda x:A.\,t\)
        & \(t\,u\) \\[4pt]
      type functions
        & \(\lambda X.\,t\)
        & \(t\,A\)
    \end{tabular}
  \end{center}

  \vfill

  A term of type
  \[
    \forall X.\,A
  \]
  is a program that can be instantiated at any type \(B\):
  \[
    t\,B : A\subst{B}{X}.
  \]
\end{frame}

\begin{frame}{Types}
  Fix a countably infinite set $\mathbb{V}$ of type variables.  A type context
  $\Delta$ is a finite sequence of distinct type variables.

  \vfill

  \begin{definition}
    The judgement $\Delta \vdash A : \type$ is defined inductively as follows.
    \begin{mathpar}
      \inferrule*[right=\textsc{TVar}]
        {X \in \Delta}
        {\Delta \vdash X : \type}

      \inferrule*[right=\textsc{TFun}]
        {\Delta \vdash A : \type \\ \Delta \vdash B : \type}
        {\Delta \vdash A \to B : \type}

      \inferrule*[right=\textsc{TAll}]
        {\Delta, X \vdash A : \type}
        {\Delta \vdash \forall X.\,A : \type}
    \end{mathpar}
  \end{definition}

  The type variable $X$ is bound in $\forall X.\,A$.
\end{frame}

\begin{frame}{Free type variables and substitution}
  Free type variables are computed by
  \begin{align*}
    \FV(X)              & = \{X\}, \\
    \FV(A \to B)        & = \FV(A) \cup \FV(B), \\
    \FV(\forall X.\,A)  & = \FV(A) - \{X\}.
  \end{align*}

  \vfill

  Type substitution $A\subst{B}{X}$ means substituting the type $B$ for the
  free occurrences of $X$ in $A$, avoiding capture by bound type variables.

  \begin{example}
    \[
      (\forall Y.\,X \to Y)\subst{B}{X}
      =
      \forall Y.\,B \to Y
    \]
    provided that $Y \notin \FV(B)$.
  \end{example}
\end{frame}

\begin{frame}{Terms}
  \begin{definition}
    The terms of System~F extend simply typed terms with explicit type
    abstraction and type application:
    \begin{description}
      \item[variable] $x \in \Lambda_{\forall}(V,\mathbb{V})$ if $x \in V$;
      \item[application] $t@u \in \Lambda_{\forall}(V,\mathbb{V})$ if
        $t,u \in \Lambda_{\forall}(V,\mathbb{V})$;
      \item[abstraction] $\lam{x:A}t \in \Lambda_{\forall}(V,\mathbb{V})$ if
        $x \in V$, $A$ is a type, and $t \in \Lambda_{\forall}(V,\mathbb{V})$;
      \item[type abstraction] $\lam{X}t \in \Lambda_{\forall}(V,\mathbb{V})$ if
        $X \in \mathbb{V}$ and $t \in \Lambda_{\forall}(V,\mathbb{V})$;
      \item[type application] $t\;A \in \Lambda_{\forall}(V,\mathbb{V})$ if
        $t \in \Lambda_{\forall}(V,\mathbb{V})$ and $A$ is a type.
    \end{description}
  \end{definition}

  \vfill

  Term abstraction abstracts over a term variable:
  \[
    \lam{x : A}t.
  \]
  Type abstraction abstracts over a type variable:
  \[
    \lam{X}t.
  \]
\end{frame}

\begin{frame}{Contexts and judgements}
  System~F has two contexts.

  \begin{description}
    \item[Type context]
      \[
        \Delta ::= \cdot \mid \Delta, X
      \]
      records available type variables.

    \item[Term context]
      \[
        \Gamma ::= \cdot \mid \Gamma, x : A
      \]
      records available term variables and their types.
  \end{description}

  \vfill

  The main judgements are
  \[
    \Delta \vdash A : \type
    \qquad\text{and}\qquad
    \Delta;\Gamma \vdash t : A.
  \]

  The judgement $\vdash t : A$ abbreviates $\cdot;\cdot \vdash t : A$.
\end{frame}

\begin{frame}{Typing rules}
  The typing rules for variables, abstraction, and application are inherited
  from simply typed $\lambda$-calculus, with the additional type context.

  \begin{prooftree}
    \AXC{}
    \RightLabel{if $\Gamma \ni x : A$}
    \UIC{$\Delta;\Gamma \vdash x : A$}
  \end{prooftree}

  \begin{multicols}{2}
    \begin{prooftree}
      \AXC{$\Delta \vdash A : \type$}
      \AXC{$\Delta;\Gamma,x:A \vdash t : B$}
      \BIC{$\Delta;\Gamma \vdash \lam{x : A}t : A \to B$}
    \end{prooftree}

    \begin{prooftree}
      \AXC{$\Delta;\Gamma \vdash t : A \to B$}
      \AXC{$\Delta;\Gamma \vdash u : A$}
      \BIC{$\Delta;\Gamma \vdash t\;u : B$}
    \end{prooftree}
  \end{multicols}
\end{frame}

\begin{frame}{Polymorphic typing rules}
  The new rules introduce and eliminate universal types.

  \begin{prooftree}
    \AXC{$\Delta,X;\Gamma \vdash t : A$}
    \RightLabel{$\forall$-intro}
    \UIC{$\Delta;\Gamma \vdash \lam{X}t : \forall X.\,A$}
  \end{prooftree}

  \vfill

  \begin{prooftree}
    \AXC{$\Delta;\Gamma \vdash t : \forall X.\,A$}
    \AXC{$\Delta \vdash B : \type$}
    \RightLabel{$\forall$-elim}
    \BIC{$\Delta;\Gamma \vdash t\;B : A\subst{B}{X}$}
  \end{prooftree}

  \vfill

  The side condition in $\forall$-introduction is carried by the type context:
  any occurrence of $X$ in $A$ is bound by the introduced quantifier.
\end{frame}

\begin{frame}{Example: polymorphic identity}
  The polymorphic identity function is $\mathtt{id} \defeq \lam{X}\lam{x : X}x$.

  Its type is derivable as follows.
  \begin{prooftree}
    \AXC{}
    \UIC{$X \vdash X : \type$}
    \AXC{}
    \UIC{$X;x:X \vdash x : X$}
    \BIC{$X;\cdot \vdash \lam{x : X}x : X \to X$}
    \UIC{$\cdot;\cdot \vdash \lam{X}\lam{x : X}x : \forall X.\,X \to X$}
  \end{prooftree}

  Instantiation at a type $A$ gives $\mathtt{id}\;A : A \to A$.
  \vfill

  \begin{exercise*}
    Reduce the following term:
    \[
      \mathtt{id}\;(A \to A)\;(\lam{x:A}x).
    \]

    What is its type?
  \end{exercise*}
\end{frame}


\begin{frame}{Exercises}
  Derive the following judgements.

  \begin{enumerate}
    \item
      \[
        \vdash \lam{X}\lam{Y}\lam{x : X}\lam{y : Y}x
        : \forall X.\,\forall Y.\,X \to Y \to X.
      \]

    \item
      \[
        \vdash
        \lam{X}\lam{f : X \to X}\lam{x : X}f\;(f\;x)
        : \forall X.\,(X \to X) \to X \to X.
      \]

    \item Explain why a term of type $\forall X.\,X \to X$ cannot use any
      operation specific to a particular type.
  \end{enumerate}
\end{frame}

\section{Polymorphic \texorpdfstring{$\lambda$}{λ}-Calculus: Dynamics and Programming}

\begin{frame}{Reduction}
  System~F has the ordinary term-level $\beta$-rule and a type-level
  $\beta$-rule.

  \begin{align*}
    (\lam{x : A}t)\;u
      &\onereduce t\subst{u}{x}, \\
    (\lam{X}t)\;A
      &\onereduce t\subst{A}{X}.
  \end{align*}

  \vfill

  Example:
  \begin{align*}
    (\lam{X}\lam{x : X}x)\;A\;u
      &\onereduce (\lam{x : X}x)\subst{A}{X}\;u \\
      &\equiv (\lam{x : A}x)\;u \\
      &\onereduce u.
  \end{align*}
\end{frame}

\begin{frame}{Type safety and normalisation}
  The same structural proof pattern as in simply typed $\lambda$-calculus
  extends to System~F.

  \begin{theorem}[Type safety]
    If $\Delta;\Gamma \vdash t : A$ and $t \onereduce u$, then
    $\Delta;\Gamma \vdash u : A$.
  \end{theorem}

  \begin{theorem}[Progress]
    If $\cdot;\cdot \vdash t : A$, then either $t$ is normal or there exists
    $u$ such that $t \onereduce u$.
  \end{theorem}

  \vfill

  System~F also enjoys strong normalisation: every well-typed reduction
  sequence terminates.  We state this theorem without proof.
\end{frame}

\begin{frame}{Sum type}
  A binary sum can be encoded by asking the sum value how to eliminate itself.

  \begin{definition}
    \[
      A + B \defeq \forall X.\,(A \to X) \to (B \to X) \to X.
    \]
  \end{definition}
  Read this as:

  \begin{quote}
    To use an \(A+B\), give a result type \(X\), a case for \(A\), and a case
    for \(B\).
  \end{quote}

  \vfill

  The injections are
  \begin{align*}
    \mathtt{left}_{A,B}
      &\defeq
        \lam{x : A}\lam{X}\lam{f : A \to X}\lam{g : B \to X}f\;x, \\
    \mathtt{right}_{A,B}
      &\defeq
        \lam{y : B}\lam{X}\lam{f : A \to X}\lam{g : B \to X}g\;y.
  \end{align*}

  \begin{exercise*}
    Define
    \[
      \mathtt{either} : \forall X.\,(A \to X) \to (B \to X) \to A + B \to X.
    \]
  \end{exercise*}
\end{frame}

\begin{frame}{Empty type}
  The polymorphic type
  \[
    \bot \defeq \forall X.\,X
  \]
  behaves as an empty type: a closed term of this type would have to produce an
  element of every type.

  \vfill

  \begin{exercise*}
    Think informally about what a closed term of type $\forall X.\,X$ would
    have to do.

    \begin{itemize}
      \item It must work after $X$ is instantiated with any type.
      \item It receives no term argument from which an $X$ could be obtained.
      \item So where could a value of the chosen type come from?
    \end{itemize}
  \end{exercise*}

  We will return to a parametricity argument for non-existence later.
\end{frame}

\begin{frame}{Product type}
  A binary product can be encoded by asking the pair how to eliminate itself.

  \begin{definition}
    \[
      A \times B \defeq \forall X.\,(A \to B \to X) \to X.
    \]
  \end{definition}

  \vfill

  Pairing is
  \[
    \langle -, - \rangle_{A,B}
      \defeq
      \lam{x : A}\lam{y : B}\lam{X}\lam{f : A \to B \to X}f\;x\;y.
  \]

\end{frame}

\begin{frame}{Exercise}
  Define
  \[
    \mathtt{split}
    :
    A \times B \to \forall X.\,(A \to B \to X) \to X.
  \]

  Then use it to define projections
  \[
    \mathtt{proj}_1 : A \times B \to A
    \qquad\text{and}\qquad
    \mathtt{proj}_2 : A \times B \to B.
  \]
\end{frame}


\begin{frame}[allowframebreaks]{Natural numbers}
  Church numerals become a single closed polymorphic type:
  \[
    \nat \defeq \forall X.\,(X \to X) \to X \to X.
  \]

  \begin{description}
    \item[Numerals]
      \[
        \bc_n \defeq \lam{X}\lam{f : X \to X}\lam{x : X}f^n\;x.
      \]

    \item[Successor]
      \[
        \suc \defeq
        \lam{n : \nat}\lam{X}\lam{f : X \to X}\lam{x : X}
          f\;(n\;X\;f\;x).
      \]

    \item[Fold]
      \[
        \fold_{\nat}
        \defeq
        \lam{X}\lam{f : X \to X}\lam{x : X}\lam{n : \nat}n\;X\;f\;x.
      \]
  \end{description}

  \framebreak

  Addition and multiplication can be written directly:
  \begin{align*}
    \add
      &\defeq
        \lam{m : \nat}\lam{n : \nat}\lam{X}\lam{f : X \to X}\lam{x : X}
          m\;X\;f\;(n\;X\;f\;x), \\
    \mul
      &\defeq
        \lam{m : \nat}\lam{n : \nat}\lam{X}\lam{f : X \to X}\lam{x : X}
          m\;X\;(n\;X\;f)\;x.
  \end{align*}

  \begin{exercise*}
    Rewrite $\add$ and $\mul$ using $\fold_{\nat}$.
  \end{exercise*}
\end{frame}

\begin{frame}{Lists}
  Lists over a type $A$ are encoded by their fold operator.

  \begin{definition}
    \[
      \List(A) \defeq \forall X.\,X \to (A \to X \to X) \to X.
    \]
  \end{definition}

  \vfill

  Constructors:
  \begin{align*}
    \mathtt{nil}_A
      &\defeq
        \lam{X}\lam{z : X}\lam{c : A \to X \to X}z, \\
    \mathtt{cons}_A
      &\defeq
        \lam{x : A}\lam{xs : \List(A)}\lam{X}\lam{z : X}\lam{c : A \to X \to X}
          c\;x\;(xs\;X\;z\;c).
  \end{align*}

  \begin{exercise*}
    Define
    \[
      \mathtt{map} : (A \to B) \to \List(A) \to \List(B).
    \]
  \end{exercise*}
\end{frame}

\section{Reasoning with Types}

\begin{frame}{What can types tell?}
  Types rule out bad programs, but polymorphic types can also rule out many
  seemingly possible behaviours.

  In fact, types can be used to tell which functions are definable, and which
  equations a term should satisfy at a given type.

  \vfill

  For example, what can a closed term of each type do?
  \begin{enumerate}
    \item $\forall X.\,X$
    \item $\forall X.\,X \to X$
    \item $\forall X.\,\forall Y.\,X \to Y \to X$
    \item $\forall X.\,X \to \nat$
  \end{enumerate}

  \vfill

  The informal answer is: a polymorphic program must act uniformly in the
  unknown type.  We first look at functions definable in simply typed
  $\lambda$-calculus.
\end{frame}

\begin{frame}[allowframebreaks]{$\lambda$-definability in simply typed $\lambda$-calculus}
  Each term $\Gamma \vdash t : A$ can be interpreted as a
  set-theoretic function
  \[
    \Brackets{t} : \Brackets{\Gamma} \to \Brackets{A}.
  \]

  Assign a set $O_X$ to each type variable $X \in \mathbb{V}$, then extend it to all types:
  \begin{align*}
    \Brackets{X}       & = O_X, \\
    \Brackets{A \to B} & = \Brackets{A} \to \Brackets{B},
  \end{align*}
  and to contexts:
  \begin{align*}
    \Brackets{\cdot}         & = \{*\}, \\
    \Brackets{\Gamma, x : A} & = \Brackets{\Gamma} \times \Brackets{A}.
  \end{align*}

  \framebreak
  The interpretation of terms is defined inductively, modulo $\alpha$-equivalence:
  \begin{align*}
    \Brackets{\Gamma \vdash x_i : A}(\rho)
      & = \rho(i), \\
    \Brackets{\Gamma \vdash t\;u : B}(\rho)
      & = \Brackets{\Gamma \vdash t : A \to B}(\rho)
          \left(\Brackets{\Gamma \vdash u : A}(\rho)\right), \\
    \Brackets{\Gamma \vdash \lam{x}t : A \to B}(\rho)
      & = \left(v \mapsto \Brackets{\Gamma, x : A \vdash t : B}(\rho,v)\right).
  \end{align*}
  Here $\rho \in \Brackets{\Gamma}$ is called an \emph{environment}.
  For $\Brackets{\cdot \vdash t : A}(*)$, we simply write $\Brackets{t}$.

  \begin{definition}
    A set-theoretic function
    $f : \Brackets{A} \to \Brackets{B}$ is \alert{$\lambda$-definable} at type
    $A \to B$ if there is a closed term $t : A \to B$ such that
    $f = \Brackets{t}$.
  \end{definition}
\end{frame}

\begin{frame}{Definability}
  Let there be one base type variable $X$, interpreted by
  \[
    O_X = \{\btt,\bff\}.
  \]

  The semantic function space contains four functions $O_X \to O_X$:
  \begin{enumerate}
    \item identity: $a \mapsto a$;
    \item constant true: $a \mapsto \btt$;
    \item constant false: $a \mapsto \bff$;
    \item negation: $\btt \mapsto \bff$ and $\bff \mapsto \btt$.
  \end{enumerate}

  \vfill

  The question is not whether these set-theoretic functions exist.  They do.
  The question is which of them are denotations of closed $\lambda$-terms of
  type $X \to X$.
\end{frame}

\begin{frame}{Why relations?}
  The set-theoretic interpretation tells us which functions exist.

  \vfill

  But we want to know which functions are \emph{definable} by programs.

  \vfill

  Logical relations give a test:

  \begin{quote}
    A definable function must preserve every relation compatible with its type.
  \end{quote}

  \vfill

  To show that a set-theoretic function is not definable, find a relation that
  it fails to preserve.
\end{frame}

\begin{frame}{Logical relation}
  \begin{idea}
    If $v_1$ and $v_2$ are related, then a definable function must send them to
    related outputs.
  \end{idea}

  A family $\{R^A \subseteq \Brackets{A} \times \Brackets{A}\}_{A : \type}$
  of binary relations is \alert{logical} if
  \[
    R^{A \to B}(f_1,f_2)
    \quad\text{iff}\quad
    \forall x_1\,x_2.\,
      R^{A}(x_1,x_2) \implies R^{B}(f_1(x_1),f_2(x_2)).
  \]
  \vfil

  N.B.\ A logical relation is determined by $R^X$ for type variables $X$.
\end{frame}

\begin{frame}{Exercise}

    What is $R^{X \to X}$ if \dots
    \begin{enumerate}
      \item $R^X = \emptyset$?
      \item $R^X = O_X \times O_X$?
      \item $R^X = \{(\btt,\bff)\}$?
    \end{enumerate}

    Let \(O_X=\{\btt,\bff\}\) and $R^X = \{(\bff,\btt)\}$.

    Which of the following functions preserve \(R^X\)?

    \begin{enumerate}
      \item identity;
      \item constant true;
      \item constant false;
      \item negation.
    \end{enumerate}
\end{frame}


\begin{frame}{The fundamental theorem of logical relations}
  \begin{theorem}[Fundamental Theorem of Logical Relations]\label{thm:fundamental_thm}
    Let $\{R^A\}_{A : \type}$ be a logical relation.  For every typing
    derivation $\Gamma \vdash t : A$ and every pair of environments
    $\rho_1,\rho_2 \in \Brackets{\Gamma}$ that are pointwise related, we have
    \[
      R^A(\Brackets{\Gamma \vdash t : A}(\rho_1),
          \Brackets{\Gamma \vdash t : A}(\rho_2)).
    \]
    Pointwise related means that $R^{A_i}(\rho_1(i),\rho_2(i))$ for every
    $x_i : A_i \in \Gamma$.
  \end{theorem}

  \begin{proof}[Proof sketch]
    By induction on the typing derivation of $\Gamma \vdash t : A$.
  \end{proof}

  In particular, $R^A(\Brackets{t},\Brackets{t})$ for any closed term $t$ of
  type $A$.
\end{frame}

\begin{frame}{Definability, revisited}
  Consider $O_X = \{\btt,\bff\}$ and the logical relation determined by
  \[
    R^{X} = \{(\bff,\btt)\}.
  \]

  \begin{proof}
    Suppose that the constant true function $c_{\btt}(x)=\btt$ is
    $\lambda$-definable.  Then there is a closed term $t : X \to X$ with
    $\Brackets{t}=c_{\btt}$.

    By the fundamental theorem,
    \[
      R^{X \to X}(\Brackets{t},\Brackets{t}).
    \]
    Since $R^X(\bff,\btt)$, the arrow clause gives
    \[
      R^X(\Brackets{t}(\bff),\Brackets{t}(\btt)).
    \]
    But $\Brackets{t}$ is constant true, so this says
    $R^X(\btt,\btt)$, a contradiction.
  \end{proof}
\end{frame}

\begin{frame}{Exercises}
  \begin{enumerate}
    \item Show that the constant function $f(x) = \bff$ is not $\lambda$-definable.
    \item Show that the negation function $\neg$ is not $\lambda$-definable.
  \end{enumerate}
\end{frame}

\begin{frame}{From logical relations to parametricity}
  For simply typed terms, logical relations tell us:

  \[
    \text{definable programs preserve relations.}
  \]

  \vfill

  For polymorphic terms, the relation for a type variable is not fixed.

  \[
    X \quad\leadsto\quad \text{any relation chosen by the user}
  \]

  \vfill

  Parametricity says that a polymorphic program must work uniformly for every
  such choice.
\end{frame}

\begin{frame}{No set-theoretic model for polymorphic $\lambda$-calculus}
  We would like to apply the same approach to polymorphic $\lambda$-calculus,
  but the impredicative universal type makes the naive set-theoretic account
  circular:
  \begin{enumerate}
    \item the universal quantification $\forall X.\,A$ is \alert{impredicative};
    \item $\Brackets{\forall X.\,A}$ should depend on
      $\Brackets{A\subst{B}{X}}$ for every $B : \type$;
    \item this includes the case $B = \forall X.\,A$ itself.
  \end{enumerate}

  In fact, Reynolds showed that polymorphism is not set-theoretic in classical
  set theory, due to a cardinality obstruction~\cite{Reynolds1984}.

  Thus one has to use other models, constructive set theory~\cite{Pitts1987},
  or weaker predicative versions of parametric polymorphism~\cite{Leivant1991}.
\end{frame}

\begin{frame}{Parametricity in polymorphic $\lambda$-calculus}
  Following Girard's \emph{reducibility candidates}~\cite{Girard1989}, assume a
  set $\mathcal{U}$ of \alert{relation candidates} in some model.
  A family $\{R_{\Phi}^{A}\}_{\Delta \vdash A}$ is logical if
  \begin{align*}
    R^{X}_{\Phi}(x_1,x_2)
      & \quad\text{iff}\quad \Phi(X)(x_1,x_2), \\
    R^{A \to B}_{\Phi}(f_1,f_2)
      & \quad\text{iff}\quad
      \forall x_1\,x_2.\,
        R^{A}_{\Phi}(x_1,x_2) \implies
        R^{B}_{\Phi}(f_1(x_1),f_2(x_2)), \\
    R^{\forall X.\,A}_{\Phi}(p_1,p_2)
      & \quad\text{iff}\quad
      \alert{\forall U \in \mathcal{U}.\,
        R^{A}_{\Phi; X \mapsto U}(p_1(U),p_2(U))}.
  \end{align*}
  Here $\Phi\colon \Delta \to \mathcal{U}$ maps type variables to relation
  candidates, and $\Phi;X \mapsto U$ maps $X$ to $U$ and maps every other
  variable $Y$ to $\Phi(Y)$.

  If $\Delta$ is empty, the subscript $\Phi$ in $R^A_{\Phi}$ is omitted.
  \begin{theorem}
    The fundamental theorem holds for these logical relations.  In particular,
    $R^A(\Brackets{t},\Brackets{t})$ holds for every closed polymorphic term
    $t : A$.
  \end{theorem}
\end{frame}

\begin{frame}{Example: $\forall X.\,X$}
  The type $\forall X.\,X$ is not inhabited.

  Suppose that $\vdash t : \forall X.\,X$.
  Then, by the fundamental theorem,
  \[
    R^{\forall X.\,X}(\Brackets{t},\Brackets{t}).
  \]
  By the definition of the relation at a universal type, this means
  \[
    \forall U \in \mathcal{U}.\,
      R^X_{X \mapsto U}(\Brackets{t}(U),\Brackets{t}(U)),
  \]
  or equivalently,
  \[
    \forall U \in \mathcal{U}.\,
      U(\Brackets{t}(U),\Brackets{t}(U)).
  \]
  Choosing $U$ to be the empty relation gives
  \[
    (\Brackets{t}(U),\Brackets{t}(U)) \in \emptyset,
  \]
  a contradiction.  Hence there is no closed term of type $\forall X.\,X$.
\end{frame}

\begin{frame}{Exercise}
  Use parametricity to explain why every closed term of type
  \[
    \forall X.\,X \to X
  \]
  behaves like the identity.

  Hint: choose a relation containing exactly one pair \((a,b)\).
  
\end{frame}

\begin{frame}{Theorems for free}
  Instantiate the relation candidate for $X$ with the graph relation
  \[
    R^X = \{(a,f(a)) \mid a \in \Brackets{A}\}
  \]
  for some function $f\colon \Brackets{A} \to \Brackets{B}$.

  Applying the fundamental theorem yields, for any
  $t : \forall X.\,\List(X) \to \List(X)$, the commutative diagram
  \[
    \xymatrix{
      \Brackets{\List(A)} \ar[r]^{\Brackets{t}_A} \ar[d]_{\mathtt{map}\;f}
        & \Brackets{\List(A)} \ar[d]^{\mathtt{map}\;f} \\
      \Brackets{\List(B)} \ar[r]_{\Brackets{t}_B}
        & \Brackets{\List(B)}
    }
  \]

  N.B.\ The equation is derived in the working model, so it does not necessarily
  imply $=_{\beta}$ between $\lambda$-terms.

  This specialised form of the fundamental theorem is known as a
  \emph{free theorem}~\cite{Wadler1989}.
\end{frame}

\begin{frame}{Summary}
  \begin{enumerate}
    \item System~F adds type abstraction and type application to simply typed
      $\lambda$-calculus.
    \item Universal types support uniform programs such as identity, projections,
      and Church encodings.
    \item Type safety and strong normalisation still hold.
    \item Parametricity says that polymorphic programs preserve relations, which
      gives equations `for free' from types.
  \end{enumerate}

  \vfill

  This is the point at which type theory begins to act not only as a discipline
  for programs, but also as a method for reasoning about what programs can be.
\end{frame}
