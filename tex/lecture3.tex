\section{Simply Typed \texorpdfstring{$\lambda$}{λ}-Calculus: Introduction}

\begin{frame}{Adding types to a language}
  Untyped $\lambda$-calculus is expressive and computationally powerful, but it
  is difficult to read programs from their syntax alone.

  \medskip

  A function can be applied to any term: a Boolean encoding, a Church numeral,
  another function, or a divergent term.  As a programming language, the
  untyped calculus does not make the programmer's \alert{intention} explicit.

  \vfill

  We introduce a \alert{typing judgement}
  \[
    \Gamma \vdash t : A,
  \]
  meaning that the term $t$ has type $A$ under the assumptions in the context
  $\Gamma$.
\end{frame}

\section{Simply Typed \texorpdfstring{$\lambda$}{λ}-Calculus: Statics}

\begin{frame}{Types}
  Assume that $\mathbb{V}$ is a set of type variables, distinct from term
  variables.  We suppress this distinction when it is clear from context.

  \begin{definition}
    The judgement $A : \type$ is defined inductively as follows.
    \begin{multicols}{2}
      \begin{prooftree}
        \AXC{$\vphantom{X}$}
        \RightLabel{if $X \in \mathbb{V}$}
        \UIC{$X : \type$}
      \end{prooftree}
      \begin{prooftree}
        \AXC{$A : \type$}
        \AXC{$B : \type$}
        \BIC{$A \to B : \type$}
      \end{prooftree}
    \end{multicols}
  \end{definition}

  We say that $A$ is a type if $A : \type$ is derivable.  The type
  $A \to B$ is the type of functions from $A$ to $B$.
\end{frame}

\begin{frame}{Higher-order function types}
  Function types are \alert{higher-order} because
  \begin{enumerate}
    \item functions can be arguments of other functions;
    \item functions can be returned as results.
  \end{enumerate}

  For example,
  \begin{description}
    \item[$(A_1 \to A_2) \to B$]
      is the type of functions whose argument has type $A_1 \to A_2$;
    \item[$A_1 \to (A_2 \to B)$]
      is the type of functions returning a function of type $A_2 \to B$.
  \end{description}

  \vfill

  \begin{block}{Convention}
    Function types associate to the right:
    \[
      A_1 \to A_2 \to \cdots \to A_n
      \defeq
      A_1 \to (A_2 \to (\cdots \to (A_{n-1} \to A_n)\cdots)).
    \]
  \end{block}
\end{frame}

\begin{frame}{Context}
  \begin{definition}
    A \emph{typing context} $\Gamma$ is a sequence
    \[
      \Gamma \equiv x_1 : A_1,\; x_2 : A_2,\; \ldots,\; x_n : A_n
    \]
    of \alert{distinct} variables $x_i$ with types $A_i$.
  \end{definition}

  \begin{definition}
    The membership judgement $\Gamma \ni (x : A)$ is defined inductively:
    \begin{multicols}{2}
      \begin{prooftree}
        \AXC{$\vphantom{\Gamma}$}
        \UIC{$\Gamma, x : A \ni x : A$}
      \end{prooftree}
      \begin{prooftree}
        \AXC{$\Gamma \ni x : A$}
        \UIC{$\Gamma, y : B \ni x : A$}
      \end{prooftree}
    \end{multicols}
    We say that $x$ of type $A$ occurs in $\Gamma$ if
    $\Gamma \ni (x : A)$ is derivable.
  \end{definition}
\end{frame}

\begin{frame}{Exercise: types and contexts}
  Work out the following derivations.

  \begin{enumerate}
    \item Derive
      \[
        X \to Y \to X : \type.
      \]

    \item Let
      \[
        \Gamma \equiv x : A,\; y : B,\; z : A \to B.
      \]
      Which of the following membership judgements are derivable?
      \[
        \Gamma \ni x : A,
        \qquad
        \Gamma \ni y : A,
        \qquad
        \Gamma \ni z : A \to B.
      \]
  \end{enumerate}
\end{frame}

\begin{frame}{Typing rules: Curry-style system}
  The implicit typing system for simply typed $\lambda$-calculus is defined by
  the following rules.

  \begin{prooftree}
    \AXC{}
    \RightLabel{(var)\quad if $\Gamma \ni (x : A)$\quad}
    \UIC{$\Gamma \vdash_i x : A$}
  \end{prooftree}

  \begin{prooftree}
    \AXC{$\Gamma, x : A \vdash_i t : B$}
    \RightLabel{(abs)}
    \UIC{$\Gamma \vdash_i \lambda x.\,t : A \to B$}
  \end{prooftree}

  \begin{prooftree}
    \AXC{$\Gamma \vdash_i t : A \to B$}
    \AXC{$\Gamma \vdash_i u : A$}
    \RightLabel{(app)}
    \BIC{$\Gamma \vdash_i t\;u : B$}
  \end{prooftree}

  We say that $t$ is a \alert{closed typed term} if
  $\vdash_i t : A$ is derivable for some type $A$.
\end{frame}

\begin{frame}{Syntax-directedness}
  A typing system is \emph{syntax-directed} if it has exactly one typing rule
  for each term constructor.

  \vfill

  Since this typing system is syntax-directed, every typing derivation can be
  inverted.

  \begin{lemma}[Typing inversion]
    Suppose that $\Gamma \vdash_i t : A$ is derivable.
    \begin{description}
      \item[$t \equiv x$]
        implies $\Gamma \ni (x : A)$.
      \item[$t \equiv \lambda x.\,t'$]
        implies $A = B \to C$ and $\Gamma, x : B \vdash_i t' : C$ for some
        types $B$ and $C$.
      \item[$t \equiv u\;v$]
        implies there is a type $B$ such that
        $\Gamma \vdash_i u : B \to A$ and $\Gamma \vdash_i v : B$.
    \end{description}
  \end{lemma}
\end{frame}

\begin{frame}{Typing derivation}
  For any types $A$ and $B$, the judgement
  \[
    \vdash_i \lambda x\,y.\,x : A \to B \to A
  \]
  has the following derivation.

  \begin{prooftree}
    \AXC{}
    \RightLabel{(var)}
    \UIC{$x : A, y : B \vdash_i x : A$}
    \RightLabel{(abs)}
    \UIC{$x : A \vdash_i \lambda y.\,x : B \to A$}
    \RightLabel{(abs)}
    \UIC{$\vdash_i \lambda x\,y.\,x : A \to B \to A$}
  \end{prooftree}

  Therefore, $\lambda x\,y.\,x$ is a program of type $A \to B \to A$.
\end{frame}

\begin{frame}{Exercise: typing derivation}
  Derive the typing judgement
  \[
    \vdash_i
    \lambda f\,g\,x.\,f\,x\,(g\,x)
    :
    (A \to B \to C) \to (A \to B) \to A \to C
  \]
  for arbitrary types $A$, $B$, and $C$.

  \vfill

  Hint: first derive the following judgements under the context
  $f : A \to B \to C,\; g : A \to B,\; x : A$:
  \[
    f\,x : B \to C,
    \qquad
    g\,x : B.
  \]
\end{frame}

\begin{frame}{Type equations}
  When we try to assign a type to a term, the typing rules generate equality
  constraints between types.  We call these constraints \alert{type equations}.

  \vfill

  For example, to type an application $t\;u$, the function part must have a
  function type:
  \[
    \Gamma \vdash_i t : A \to B,
    \qquad
    \Gamma \vdash_i u : A.
  \]
  If $t$ has already been assigned type $C$, then we must solve
  \[
    C = A \to B.
  \]

  \vfill

  In simply typed $\lambda$-calculus, types are finite trees built from type
  variables and $\to$.  Thus an equation such as
  \[
    A = A \to B
  \]
  has no solution among simple types.
\end{frame}

\begin{frame}{Exercise: inversion and non-typability}
  Use typing inversion to decide whether the following terms are typable.

  \begin{enumerate}
    \item $\lambda x.\,x\;x$
    \item $\lambda f\,x.\,f\;x$
    \item $\lambda x\,y.\,x\;(y\;x)$
  \end{enumerate}

  \vfill

  For typable terms, give a possible type.  For non-typable terms, identify
  the unsolvable type equation that would be required.
\end{frame}

\section{Programming in Simply Typed \texorpdfstring{$\lambda$}{λ}-Calculus}

\begin{frame}[allowframebreaks]{Church encodings of natural numbers}
  For any type $A$, define the Church-numeral type at $A$ by
  \[
    \nat_A \defeq (A \to A) \to A \to A.
  \]

  \begin{description}
    \item[Church numerals]
      \begin{align*}
        \bc_n &\defeq \lambda f\,x.\,f^n x \\
        \vdash_i{}\; &\bc_n : \nat_A
      \end{align*}

    \item[Successor]
      \begin{align*}
        \suc &\defeq \lambda n\,f\,x.\,f\,(n\;f\;x) \\
        \vdash_i{}\; &\suc : \nat_A \to \nat_A
      \end{align*}

    \item[Addition]
      \begin{align*}
        \add &\defeq \lambda n\,m\,f\,x.\,m\;f\;(n\;f\;x) \\
        \vdash_i{}\; &\add : \nat_A \to \nat_A \to \nat_A
      \end{align*}

    \item[Multiplication]
      \begin{align*}
        \mul &\defeq \lambda n\,m\,f\,x.\,m\;(n\;f)\;x \\
        \vdash_i{}\; &\mul : \nat_A \to \nat_A \to \nat_A
      \end{align*}

    \item[Zero test]
      \begin{align*}
        \ifz &\defeq \lambda n\,x\,y.\,n\;(\lambda z.\,y)\;x \\
        \vdash_i{}\; &\ifz : \nat_A \to A \to A \to A
      \end{align*}
  \end{description}
\end{frame}

\begin{frame}{Exercise: typing Church programs}
  Work in the simply typed $\lambda$-calculus.

  \begin{enumerate}
    \item Show that
      \[
        \vdash_i \suc : \nat_A \to \nat_A.
      \]

    \item Show that
      \[
        \vdash_i \ifz : \nat_A \to A \to A \to A.
      \]

    \item Check that
      \[
        \ifz\;\bc_0\;t\;u \reduce t,
        \qquad
        \ifz\;\bc_{n+1}\;t\;u \reduce u.
      \]
  \end{enumerate}
\end{frame}

\begin{frame}{Church encodings of Boolean values}
  For any type $A$, define
  \[
    \bool_A \defeq A \to A \to A.
  \]

  \begin{description}
    \item[Boolean values]
      \[
        \true \defeq \lambda x\,y.\,x
        \qquad\text{and}\qquad
        \false \defeq \lambda x\,y.\,y.
      \]

    \item[Conditional]
      \begin{align*}
        \cond &\defeq \lambda b\,x\,y.\,b\,x\,y \\
        \vdash_i{}\; &\cond : \bool_A \to A \to A \to A
      \end{align*}
  \end{description}
\end{frame}

\begin{frame}{Exercise: typed Boolean operations}
  Define conjunction, disjunction, and negation in simply typed $\lambda$-calculus.

  \begin{enumerate}
    \item Give terms for
      \[
        \mathtt{and},\qquad \mathtt{or},\qquad \mathtt{not}.
      \]

    \item Check that the following judgements are derivable:
      \[
        \vdash_i \mathtt{and} : \bool_A \to \bool_A \to \bool_A,
      \]
      \[
        \vdash_i \mathtt{or} : \bool_A \to \bool_A \to \bool_A,
        \qquad
        \vdash_i \mathtt{not} : \bool_A \to \bool_A.
      \]
  \end{enumerate}
\end{frame}

\section{Type Safety}

\begin{frame}[c]{Type safety = preservation + progress}
  \begin{quote}
    ``Well-typed programs cannot go wrong.''\\
    \hfill -- Milner, 1978
  \end{quote}

  \begin{description}
    \item[Preservation]
      If $\Gamma \vdash_i t : A$ and $t \onereduce u$, then
      $\Gamma \vdash_i u : A$.

    \item[Progress]
      If $\Gamma \vdash_i t : A$, then either $t$ is in normal form or there
      exists $u$ such that $t \onereduce u$.
  \end{description}

  \vfill

  Preservation says that reduction does not change types.  Progress says that a
  well-typed term is never stuck except at a normal form.
\end{frame}

\begin{frame}{Converse of preservation}
  The converse of preservation does not hold: a reduct may be typable even when
  the original term is not.

  \begin{lemma}[Typability of subterms]
    If $\Gamma \vdash_i t : A$ is derivable, then every subterm of $t$ is
    typable under some context.
  \end{lemma}

  Let
  \[
    \mathbf{K}_1 \defeq \lambda x\,y.\,x,
    \qquad
    \mathbf{I} \defeq \lambda x.\,x,
    \qquad
    \Omega \defeq (\lambda x.\,x\,x)\,(\lambda x.\,x\,x).
  \]
  Then
  \[
    \mathbf{K}_1\;\mathbf{I}\;\Omega \reduce \mathbf{I}.
  \]
  The reduct $\mathbf{I}$ is typable, but $\Omega$ is not typable, so
  $\mathbf{K}_1\;\mathbf{I}\;\Omega$ is not typable.
\end{frame}

\begin{frame}{Preservation theorem}
  We use two standard lemmas.

  \begin{description}
    \item[Weakening]
      If $\Gamma \vdash_i t : A$ and $x \notin \mathrm{dom}(\Gamma)$, then
      $\Gamma, x : B \vdash_i t : A$.

    \item[Substitution]
      If $\Gamma, x : A \vdash_i t : B$ and $\Gamma \vdash_i u : A$, then
      $\Gamma \vdash_i t\subst{u}{x} : B$.
  \end{description}

  \begin{theorem}[Preservation]
    If $\Gamma \vdash_i t : A$ and $t \onereduce u$, then
    $\Gamma \vdash_i u : A$.
  \end{theorem}

  \begin{proof}[Proof sketch]
    By induction on the derivations of $\Gamma \vdash_i t : A$ and
    $t \onereduce u$.  The only interesting case is the $\beta$-redex
    $(\lambda x.\,t')\;u'$, where the substitution lemma is needed.
  \end{proof}
\end{frame}

\begin{frame}{Exercise: preservation for the $\beta$-case}
  Fill in the main case of the preservation proof.

  Suppose
  \[
    \Gamma \vdash_i (\lambda x.\,t)\;u : B
    \qquad\text{and}\qquad
    (\lambda x.\,t)\;u \onereduce t\subst{u}{x}.
  \]

  \begin{enumerate}
    \item Use inversion to obtain a type $A$ such that
      \[
        \Gamma \vdash_i \lambda x.\,t : A \to B,
        \qquad
        \Gamma \vdash_i u : A.
      \]
    \item Use inversion again to obtain
      \[
        \Gamma, x : A \vdash_i t : B.
      \]
    \item Apply substitution to conclude
      \[
        \Gamma \vdash_i t\subst{u}{x} : B.
      \]
  \end{enumerate}
\end{frame}

\begin{frame}{Normal and neutral forms}
  To prove progress constructively, we use a syntactic characterisation of
  normal forms.

  \begin{multicols}{2}
    \begin{prooftree}
      \AXC{$\vphantom{x}$}
      \UIC{$\neutral\;x$}
    \end{prooftree}

    \begin{prooftree}
      \AXC{$\neutral\;t$}
      \AXC{$\normal\;u$}
      \BIC{$\neutral\;(t\;u)$}
    \end{prooftree}

    \columnbreak

    \begin{prooftree}
      \AXC{$\neutral\;t$}
      \UIC{$\normal\;t$}
    \end{prooftree}

    \begin{prooftree}
      \AXC{$\normal\;u$}
      \UIC{$\normal\;(\lambda x.\,u)$}
    \end{prooftree}
  \end{multicols}

  \vfill

  Neutral terms are variables applied to normal arguments.  Normal terms are
  either neutral terms or abstractions whose bodies are normal.
\end{frame}

\begin{frame}{Exercise: normal and neutral forms}
  Decide which of the following terms are neutral, normal, or reducible.

  \begin{enumerate}
    \item $x$
    \item $x\;((\lambda y.\,y)\;z)$
    \item $\lambda x.\,x\;y$
    \item $(\lambda x.\,x)\;y$
    \item $x\;(\lambda y.\,y)$
  \end{enumerate}

  \vfill

  For each normal term, give a derivation of $\normal\;t$ or $\neutral\;t$.
\end{frame}

\begin{frame}{Normal-form characterisation}
  \begin{lemma}[Normal-form characterisation]
    A term $t$ has no $\beta$-reduction if and only if $\normal\;t$ is derivable.
  \end{lemma}

  \vfill

  This lemma says that the inductive judgements $\normal$ and $\neutral$ exactly
  capture the ordinary notion of having no $\beta$-redex.

  \medskip

  We will use this characterisation in the proof of progress, but we will not
  prove the lemma here.
\end{frame}

\begin{frame}{Progress theorem}
  \begin{theorem}[Progress]
    If $\Gamma \vdash_i t : A$, then $\normal\;t$ or there exists $u$ such that
    $t \onereduce u$.
  \end{theorem}

  \begin{proof}[Proof sketch]
    By induction on the derivation of $\Gamma \vdash_i t : A$.

    \medskip

    The variable and abstraction cases produce normal forms.  The application
    case is the main case: if the function part reduces, the whole application
    reduces; if the function part is an abstraction, there is a $\beta$-redex;
    otherwise the application is neutral when its argument is normal.
  \end{proof}
\end{frame}

\begin{frame}{Exercise: progress for applications}
  Fill in the application case of the progress proof.

  Suppose
  \[
    \Gamma \vdash_i t\;u : B.
  \]
  By inversion, for some type $A$,
  \[
    \Gamma \vdash_i t : A \to B,
    \qquad
    \Gamma \vdash_i u : A.
  \]

  Use the induction hypotheses for $t$ and $u$ to decide whether $t\;u$ is
  normal or reducible.  Pay special attention to the case where $t$ is an
  abstraction.
\end{frame}

\begin{frame}{Type safety: summary}
  Preservation and progress are proved separately, but they work together.

  \begin{description}
    \item[Preservation]
      Each reduction step keeps the same type.

    \item[Progress]
      A well-typed term is either already normal or can take a reduction step.
  \end{description}

  \vfill

  Hence evaluation of a well-typed term cannot get stuck because of a type
  error.  It either continues reducing while preserving its type, or it reaches
  a normal form.
\end{frame}

\begin{frame}{Strong normalisation}
  \begin{definition}
    A term $t$ is \emph{strongly normalising}, written $t \Downarrow$, if every
    reduction sequence starting from $t$ terminates.
  \end{definition}

  \begin{theorem}[Strong normalisation]
    Every typable term $t$ with $\Gamma \vdash_i t : A$ is strongly normalising.
  \end{theorem}

  \vfill

  This theorem is stronger than type safety.  Type safety says that reduction
  does not get stuck; strong normalisation says that, in the simply typed
  $\lambda$-calculus, reduction cannot continue forever.

  \medskip

  We state this result without proof.
\end{frame}

\section{Optional Extensions}

\begin{frame}{General recursion: statics}
  The self-applicative term
  \[
    \lambda x.\,x\;x
  \]
  is not typable in simply typed $\lambda$-calculus: if $x : A$, then using
  $x$ as a function applied to itself would require $A = A \to B$ for some
  type $B$.

  Hence the untyped $Y$ combinator is not typable.

  \vfill

  We can add general recursion explicitly by extending the term grammar with
  \[
    \fix\,f.\,t.
  \]

  \begin{prooftree}
    \AXC{$\Gamma, f : A \vdash_i t : A$}
    \UIC{$\Gamma \vdash_i \fix\,f.\,t : A$}
  \end{prooftree}
\end{frame}

\begin{frame}{General recursion: dynamics}
  Reduction is extended with the rule
  \[
    \fix\,f.\,t \onereduce t\subst{\fix\,f.\,t}{f}.
  \]

  \vfill

  Non-terminating well-typed terms are now easy to define.  For any type $A$,
  \[
    \vdash_i \fix\,x.\,x : A.
  \]
  Its reduction unfolds forever:
  \begin{align*}
    \fix\,x.\,x
      &\onereduce x\subst{\fix\,x.\,x}{x} \\
      &\equiv \fix\,x.\,x \\
      &\onereduce \cdots
  \end{align*}

  Adding general recursion therefore destroys strong normalisation.
\end{frame}

\begin{frame}{Exercise: fixed points and type safety}
  Work in the calculus extended with $\fix$.

  \begin{enumerate}
    \item Derive
      \[
        \vdash_i \fix\,x.\,x : A.
      \]

    \item Evaluate the first three reduction steps of
      \[
        \fix\,x.\,x.
      \]

    \item Does preservation still hold for the $\fix$ reduction rule?
      Explain which typing rule is used.

    \item Does strong normalisation still hold?
  \end{enumerate}
\end{frame}

\begin{frame}{Natural numbers: statics}
  Church numerals have many possible types $\nat_A$.  Instead, we can extend
  the calculus with a single primitive type of natural numbers.

  Add terms:
  \begin{itemize}
    \item $\zero$;
    \item $\suc(t)$, if $t$ is a term;
    \item $\ifz(t; x.\,u; v)$, if $t$, $u$, and $v$ are terms.
  \end{itemize}

  Add typing rules:
  \begin{mathpar}
    \inferrule{ }{\Gamma \vdash_i \zero : \mathbb{N}} \and
    \inferrule{\Gamma \vdash_i t : \mathbb{N}}{\Gamma \vdash_i \suc(t) : \mathbb{N}} \and
    \inferrule{\Gamma \vdash_i v : \mathbb{N} \\ \Gamma \vdash_i t : A \\ \Gamma, x : \mathbb{N} \vdash_i u : A}{\Gamma \vdash_i \ifz(t; x.\,u; v) : A}
  \end{mathpar}

  The third rule is pattern matching on natural numbers.
\end{frame}

\begin{frame}{Natural numbers: dynamics}
  Reduction for natural numbers is extended with two rules:
  \begin{align*}
    \ifz(t; x.\,u; \zero) &\onereduce t \\
    \ifz(t; x.\,u; \suc(n)) &\onereduce u\subst{n}{x}.
  \end{align*}

  \vfill

  Intuitively,
  \[
    \ifz(t; x.\,u; v)
  \]
  means: inspect $v$; if it is zero, return $t$; if it is $\suc(n)$, bind
  $n$ to $x$ and return $u$.
\end{frame}

  \]

\begin{frame}{Exercise: primitive Boolean values}

    \item What typing rules should be added?

  \]

\end{frame}
