\section{Defining data in \texorpdfstring{$\lambda$}{λ}-Calculus}

\begin{frame}{Church encoding: data by behaviour}
  A Church-encoded value is represented by what it does to its eliminators.

  \begin{center}
    \begin{tabular}{ll}
      Boolean & chooses between two branches \\
      Natural number & iterates a function a fixed number of times
    \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Church encoding of Boolean values}
  Boolean values and conditional expressions can be encoded as combinators.

  \begin{description}
    \item[Boolean values]
      \begin{align*}
        \mathtt{True}  &\defeq \lambda x\,y.\,x \\
        \mathtt{False} &\defeq \lambda x\,y.\,y
      \end{align*}
    \item[Conditional]
      \begin{align*}
        \mathtt{if} &\defeq \lambda b\;x\;y.\; b\,x\,y \\
        \mathtt{if}\;\mathtt{True}\;t\;u &\reduce t \\
        \mathtt{if}\;\mathtt{False}\;t\;u &\reduce u
      \end{align*}
      for any two $\lambda$-terms $M$ and~$N$.
  \end{description}
  where variables $x$ and $y$ of $\lambda\,x\,y$ are `used' as $\mathtt{true}$ and $\mathtt{false}$ and a Bool value is itself a conditional expression.
\end{frame}

\begin{frame}{Example}
  \begin{align*}
    \mathtt{if}\;\mathtt{True}\;t\;u
    &= (\lambda b\,x\,y.\,b\,x\,y)\;(\lambda x\,y.\,x)\;t\;u \\
    &\onereduce (\lambda x\,y.\,x)\;t\;u \\
    &\reduce t
\end{align*}
\end{frame}

\begin{frame}{Church encoding of natural numbers, I}
  Natural numbers and arithmetic operations can also be encoded.

  \begin{description}
    \item[Church numerals]
      \begin{align*}
        \bc_0     &\defeq \lambda f\,x.\,x \\
        \bc_1     &\defeq \lambda f\,x.\,f\,x \\
        \bc_2     &\defeq \lambda f\,x.\,f\,(f\,x) \\
        \bc_{n+1} &\defeq \lambda f\,x.\,f^{n+1}(x)
      \end{align*}
      where $f^1(x) \defeq f\;x$ and
      $f^{n+1}(x) \defeq f\;(f^n(x))$.
  \end{description}
  where variables $f$ and $x$ in $\bc_0$ and $\bc_{n+1}$ are used as the successor and zero under the lambda abstraction.
\end{frame}

\begin{frame}{Church encoding of natural numbers, II}
  \begin{description}
    \item[Successor]
      \begin{align*}
        \mathtt{succ} &\defeq \lambda n.\,\lambda f\,x.\, f\,(n\,f\,x) \\
        \mathtt{succ}\;\bc_n &\reduce \bc_{n+1}
      \end{align*}

    \item[Addition]
      \begin{align*}
        \add &\defeq \lambda n\,m.\,\lambda f\,x.\;
          n\;f\;(m\;f\;x) \\
        \add\;\bc_n\;\bc_m &\reduce \bc_{n+m}
      \end{align*}

    \item[Zero test]
      \begin{align*}
        \ifz &\defeq \lambda n\;x\;y.\; n\;(\lambda z.\;y)\;x \\
        \ifz\;\bc_0\;t\;u &\reduce t \\
        \ifz\;\bc_{n+1}\;t\;u &\reduce u
      \end{align*}
  \end{description}
\end{frame}

\begin{frame}{Example}
  \begin{align*}
    \mathtt{succ}\;\bc_1
    &= (\lambda n.\,\lambda f\,x.\,f\,(n\,f\,x))\;\bc_1 \\
    &\onereduce \lambda f\,x.\,f\,(\bc_1\,f\,x) \\
    &\reduce \lambda f\,x.\,f\,(f\,x) \\
    &= \bc_2.
  \end{align*}
\end{frame}


\begin{frame}{Exercise}

    \item Define the multiplication $\mul$ over Church numerals.
    \item (Hard) Define the predecessor $\mathtt{pred}$, i.e.\ $\mathtt{pred}\;\bc_{1+n} =_\beta \bc_n$.
  \end{enumerate}
\end{frame}

\section{Recursion in \texorpdfstring{$\lambda$}{λ}-Calculus}

\begin{frame}{Why direct self-reference is not a term}
  The summation $\sum_{i=0}^{n} i$, for $n \in \mathbb{N}$, is typically
  described by self-reference as
  \[
    \mathit{sum}(n) =
      \begin{cases}
        0 & \text{if } n = 0, \\
        n + \mathit{sum}(n - 1) & \text{otherwise}.
      \end{cases}
  \]
  In the pure $\lambda$-calculus, this cannot be expressed by directly naming
  $\mathit{sum}$ inside its own definition. That is, the equation
  \[
    \mathit{sum} = \lambda n.\,\ifz\;n\;\bc_0\;(\add\;n\;(\mathit{sum}\;(\pred\;n)))
  \]
  is a recursive equation at the meta-level not a $\lambda$-term.\footnote{%
  N.B. We've used $=$ for the meta-level equality but $=_\beta$ for the object-level equality.
  }
  \medskip

  A term cannot refer to its own name, but we don't have to in $\lambda$-calculus!

\begin{frame}[allowframebreaks]{General recursion as a fixpoint}
  We want to encode the recursive definition
  \[
    \mathit{sum}(n) =
      \begin{cases}
        0 & \text{if } n = 0, \\
        n + \mathit{sum}(n - 1) & \text{otherwise}.
      \end{cases}
  \]

  Instead of referring to $\mathit{sum}$ by name, define a functional $G$.
  The argument $f$ of $G\;f$ is the recursive call that the body is allowed
  to use:
  \[
    G \defeq
    \lambda f.\,
      \lambda n.\,
        \ifz\;n\;\bc_0\;
          (\add\;n\;(f\;(\pred\;n))).
  \]

  Thus $G\;f$ computes one step of summation, assuming that $f$ already knows
  how to compute the smaller recursive calls.  A genuine recursive summation
  function should therefore satisfy
  \[
    \mathtt{sum} =_\beta G(\mathtt{sum}).
  \]

  If we have a fixed-point combinator $\mathbf{Y}$ computing a fixpoint of $G$, then we will have 
  \[
    \mathtt{sum} \defeq \mathbf{Y}G,
    \qquad
    \mathbf{Y}G =_\beta G(\mathbf{Y}G).
  \]

  Therefore, in the body of $G$, the variable $f$ is instantiated by
  $\mathbf{Y}G$ itself:
  \[
    \mathbf{Y}G
    =_\beta
    \lambda n.\,
      \ifz\;n\;\bc_0\;
        (\add\;n\;((\mathbf{Y}G)\;(\pred\;n))).
  \]
\end{frame}

\begin{frame}{Curry's paradoxical combinator}

  \begin{proposition}

    The \emph{Y combinator} defined as
    \[
      \mathbf{Y} \defeq \lambda f.\,
        (\lambda x.\, f\,(x\,x))\,(\lambda x.\,f\,(x\,x)).
    \]
    is a fixed-point operator, that is for every
    $\lambda$-term $F$, $\mathbf{Y}F =_\beta F(\mathbf{Y}F)$.
  \end{proposition}
  Intuitively, $\mathbf{Y}F$ defines recursion, while $F$ describes \emph{one step} unfolding of
  the recursive definition.
  \vfil

  \begin{exercise*}
    Verify that $\mathbf{Y}F =_\beta F(\mathbf{Y}F)$ for any term $F$.
  \end{exercise*}
\end{frame}

\begin{frame}{Summation via $\mathbf{Y}$}
  We encode the recursion
  \[
    \mathit{sum}(n) =
      \begin{cases}
        0 & \text{if } n = 0, \\
        n + \mathit{sum}(n - 1) & \text{otherwise}
      \end{cases}
  \]
  by defining one iteration $G$ and $\mathtt{sum} \defeq \mathbf{Y}G$.
  Define
  \[
    G \defeq \lambda f\,n.\,
      \ifz\;n\;\bc_0\;(\add\;n\;(f\;(\pred\;n))).
  \]
  Then
  \begin{align*}
    \mathtt{sum}\;\bc_1
      &\onereduce G'\;\bc_1 \\
      &\onereduce (G\;G')\;\bc_1 \\
      &\onereduce
        (\lambda n.\,\ifz\;n\;\bc_0\;
        (\add\;n\;(G'\;(\pred\;n))))\;\bc_1 \\
      &\onereduce
        \ifz\;\bc_1\;\bc_0\;
        (\add\;\bc_1\;(G'\;(\pred\;\bc_1))) \\
      &\onereduce \cdots
  \end{align*}
  where $G' \defeq (\lambda x.\,G\;(x\;x))\;(\lambda x.\,G\;(x\;x))$.
\end{frame}

\begin{frame}{From programming to reduction theory}
  Fixed-point combinators make recursion possible in the pure
  $\lambda$-calculus, but they also expose two fundamental issues.

  \begin{description}
    \item[Non-determinacy]
      A term may contain several $\beta$-redexes.  Reducing different redexes
      first may produce different intermediate terms:
      \[
        M \reduce M_1,
        \qquad
        M \reduce M_2.
      \]

    \item[Divergence]
      Some reduction choices may unfold a recursive call forever, even when
      another choice would quickly reach a result.
  \end{description}

  \vfill

  The next two topics address these issues.

  \begin{description}
    \item[Evaluation strategies]
      How should a programming language choose which redex to reduce?

    \item[Confluence]
      If two reduction paths start from the same term, can they be joined again?
  \end{description}
\end{frame}


\section{Evaluation strategy}

\begin{frame}[allowframebreaks]{Evaluation strategies}
  An evaluation strategy is a procedure for selecting which $\beta$-redex to
  reduce. Formally, it is a subrelation
  $\longrightarrow_{\mathrm{ev}}$ of the full one-step $\beta$-reduction
  relation $\onereduce$.

  \begin{description}
    \item[Innermost $\beta$-redex] A $\beta$-redex that contains no other
      $\beta$-redex.
    \item[Outermost $\beta$-redex] A $\beta$-redex that is not contained in any
      other $\beta$-redex.
  \end{description}

  \begin{description}
    \item[Leftmost-outermost strategy (normal order)]
      Reduce the leftmost outermost $\beta$-redex first. For example,
      \begin{align*}
        &
        \underline{(\lambda x.\,(\lambda y.\,y)\,x)}\quad
        \underline{(\lambda x.\,(\lambda y.\,y\,y)\,x)} \\
        \onereduce{}
        &\underline{(\lambda y.\,y)}\quad
        \underline{(\lambda x.\,(\lambda y.\,y\,y)\,x)} \\
        \onereduce{}
        &\lambda x.\,\underline{(\lambda y.\,y\,y)}\quad
        \underline{\vphantom{(\lambda}x} \\
        \onereduce{}
        &(\lambda x.\,x\,x) \\
        &\notonereduce
      \end{align*}

    \item[Leftmost-innermost strategy]
      Reduce the leftmost innermost $\beta$-redex first. For example,
      \begin{align*}
        &(\lambda x.\,\underline{(\lambda y.\,y)}\;
          \underline{\vphantom{(\lambda}x})\,
          (\lambda x.\,(\lambda y.\,y\,y)\,x) \\
        \onereduce{}
        &(\lambda x.\,x)\;
          (\lambda x.\,\underline{(\lambda y.\,y\,y)}\;
          \underline{\vphantom{(\lambda}x}) \\
        \onereduce{}
        &\underline{(\lambda x.\,x)}\;
          \underline{(\lambda x.\,x\,x)} \\
        \onereduce{}
        &(\lambda x.\,x\,x) \\
        &\notonereduce
      \end{align*}

    \item[Rightmost-innermost and rightmost-outermost strategies]
      These are defined similarly, but choose the rightmost candidate redex
      instead.
  \end{description}
\end{frame}

\begin{frame}{CBV versus CBN}
  \begin{description}
    \item[Call-by-value strategy]
      Reduce a redex $(\lambda x.\,M)\,V$ only after its argument has been
      evaluated to a value $V$; do not reduce under abstractions.

    \item[Call-by-name strategy]
      Reduce the leftmost outermost redex, without first evaluating its
      argument; do not reduce under abstractions.
  \end{description}

  \mode<presentation>{\vfill}

  \begin{proposition}[Determinacy]
    Each fixed evaluation strategy is deterministic. That is, if
    $M \longrightarrow_{\mathrm{ev}} N_1$ and
    $M \longrightarrow_{\mathrm{ev}} N_2$, then $N_1 = N_2$.
  \end{proposition}
\end{frame}

\begin{frame}{Normalisation}
  \begin{definition}
    \begin{enumerate}
      \item $M$ is in \emph{normal form} if $M \notonereduce N$ for every
        term $N$.
      \item $M$ is \emph{weakly normalising} if $M \reduce N$ for some term
        $N$ in normal form.
    \end{enumerate}
  \end{definition}

  \[
    \Omega \defeq (\lambda x.\,x\,x)\,(\lambda x.\,x\,x),
    \qquad
    \mathbf{K}_1 \defeq \lambda x\,y.\,x.
  \]

  \begin{enumerate}
    \item $\Omega$ is not weakly normalising.
    \item $\mathbf{K}_1$ is normal, and hence weakly normalising.
    \item $\mathbf{K}_1\;z\;\Omega$ is weakly normalising.
  \end{enumerate}

  \begin{theorem}
    The normal-order strategy reduces every weakly normalising term to a normal
    form.
  \end{theorem}
\end{frame}

\section{Confluence}

\begin{frame}{Nondeterminism of $\beta$-reduction}
  Multi-step $\beta$-reduction is nondeterministic: a term may contain several
  $\beta$-redexes, and reducing different redexes first may produce different
  intermediate terms.
    \[
      \xymatrix{
        & t \ar@{->>}[rd]^(.8){*}_(.8){\beta}
            \ar@{->>}[ld]^(.8){*}_(.8){\beta} \\
        u_1 & & u_2
      }
    \]

  Confluence says that this nondeterminism is harmless: whenever two reduction
  paths start from the same term, they can be joined again by further
  reductions.
  \[
      \xymatrix{
        u_1 \ar@{->>}[rd]^(.8){*}_(.8){\beta}
          & & u_2 \ar@{->>}[ld]^(.8){*}_(.8){\beta} \\
        & v
      }
    \]
\end{frame}

\begin{frame}{Confluence}
  \begin{theorem}[Confluence]
    If $t \reduce u_1$ and $t \reduce u_2$, then there exists a term $v$ such
    that $u_1 \reduce v$ and $u_2 \reduce v$.
    \[
      \xymatrix{
        & t \ar@{->>}[rd]^(.8){*}_(.8){\beta}
            \ar@{->>}[ld]^(.8){*}_(.8){\beta} \\
        u_1 \ar@{->>}[rd]^(.8){*}_(.8){\beta}
          & & u_2 \ar@{->>}[ld]^(.8){*}_(.8){\beta} \\
        & v
      }
    \]
    The term $v$ is called a \emph{common reduct} of $u_1$ and $u_2$.
  \end{theorem}

  \vfill

  Confluence lets us reason about computational equality by reducing both sides
  to a common reduct.
  Its proof is a bit involved, so we ignore it for now.
\end{frame}

\begin{frame}{Using confluence}
  \begin{proposition}
    If $t =_\beta u$, then there exists a term $v$ such that
    $t \reduce v$ and $u \reduce v$.
  \end{proposition}

  \begin{proof}[Idea]
    The relation $=_\beta$ is generated by $\onereduce$, reflexivity, symmetry,
    and transitivity. Confluence handles the cases where reduction paths go in
    different directions.
  \end{proof}

  \vfill

  \begin{corollary}
    If $t$ and $u$ are both normal forms and $t =_\beta u$, then
    $t =_\alpha u$.
  \end{corollary}
\end{frame}

\begin{frame}{Exercise}
  Show that if $t =_\beta u$, then there is a common reduct $v$ of $t$ and $u$;
  that is, $t \reduce v$ and $u \reduce v$ by induction on the derivation of $t =_\beta u$.
\end{frame}

\section{Proof of Confluence}

\begin{frame}{Takahashi's proof of confluence}
  Proving confluence can be tricky. This section presents a direct strategy
  based on the notion of complete development, which contracts as many
  $\beta$-redexes as possible \emph{statically}.

  The \emph{complete development} $M^*$ of a $\lambda$-term $M$ is defined by
  \begin{align*}
    x^* &= x \\
    (\lambda x.\,M)^* &= \lambda x.\,M^* \\
    ((\lambda x.\,M)\;N)^* &= M^*\subst{N^*}{x} \\
    (M\;N)^* &= M^*\;N^* && \text{if $M \not\equiv \lambda x.\,M'$}.
  \end{align*}
\end{frame}

\begin{frame}{Confluence: parallel reduction}
  Let $M \parreduce N$ denote the \emph{parallel reduction} relation defined by
  the following rules.

  \begin{multicols}{2}
    \begin{prooftree}
      \AXC{$\vphantom{M\reduce M}$}
      \UIC{$x \parreduce x$}
    \end{prooftree}

    \begin{prooftree}
      \AXC{$M \parreduce N$}
      \UIC{$\lambda x.\,M \parreduce \lambda x.\,N$}
    \end{prooftree}

    \columnbreak

    \begin{prooftree}
      \AXC{$M \parreduce M'$}
      \AXC{$N \parreduce N'$}
      \BIC{$M\;N \parreduce M'\;N'$}
    \end{prooftree}

    \begin{prooftree}
      \AXC{$M \parreduce M'$}
      \AXC{$N \parreduce N'$}
      \BIC{$(\lambda x.\,M)\;N \parreduce M'\subst{N'}{x}$}
    \end{prooftree}
  \end{multicols}

  For example,
  \[
    \underline{(\lambda x.\,(\lambda y.\,y)\;x)}\;
    \underline{((\lambda x.\,x)\;\false)}
    \parreduce
    \false,
  \]
  because $(\lambda y.\,y)\;x \parreduce x$ and
  $(\lambda x.\,x)\;\false \parreduce \false$.
\end{frame}

\begin{frame}{Confluence: properties of parallel reduction}
  \begin{lemma}
    \begin{enumerate}
      \item $M \parreduce M$ holds for every term $M$,
      \item $M \onereduce N$ implies $M \parreduce N$, and
      \item $M \parreduce N$ implies $M \reduce N$.
    \end{enumerate}
    In particular, $M \parreduce^* N$ if and only if $M \reduce N$.
  \end{lemma}

  \begin{lemma}[Substitution respects parallel reduction]
    If $M \parreduce M'$ and $N \parreduce N'$, then
    $M\subst{N}{x} \parreduce M'\subst{N'}{x}$.
  \end{lemma}

  \begin{theorem}[Triangle property]
    If $M \parreduce N$, then $N \parreduce M^*$.
  \end{theorem}

  \begin{proof}[Proof sketch]
    By induction on the derivation of $M \parreduce N$.
  \end{proof}
\end{frame}

\begin{frame}{Strip lemma}
  \begin{lemma}[Strip lemma]
    If $L \parreduce^* M_1$ and $L \parreduce M_2$, then there exists a term
    $N$ such that $M_1 \parreduce N$ and $M_2 \parreduce^* N$; that is,
    \[
      \xymatrix{
        & L \ar@{=>}[rd] \ar@{=>}[ld]^(.8){*}_(.8){\beta} \\
        M_1 \ar@{=>}[rd] & & M_2 \ar@{=>}[ld]^(.8){*}_(.8){\beta} \\
        & N
      }
    \]
  \end{lemma}

  \begin{proof}[Proof sketch]
    By induction on the derivation of $L \parreduce^* M_1$.
  \end{proof}
\end{frame}

\begin{frame}{Confluence}
  \begin{theorem}
    If $L \parreduce^* M_1$ and $L \parreduce^* M_2$, then there exists a term
    $N$ such that $M_1 \parreduce^* N$ and $M_2 \parreduce^* N$.
    \[
      \xymatrix{
        & L \ar@{=>}[rd]^(.8){*}_(.8){\beta}
            \ar@{=>}[ld]^(.8){*}_(.8){\beta} \\
        M_1 \ar@{=>}[rd]^(.8){*}_(.8){\beta}
          & & M_2 \ar@{=>}[ld]^(.8){*}_(.8){\beta} \\
        & N
      }
    \]
  \end{theorem}

  \begin{corollary}
    The relation $\reduce$ is confluent.
  \end{corollary}

  This follows because $M \parreduce^* N$ is equivalent to $M \reduce N$.
\end{frame}
